// FIRESTORE SECURITY RULES
// Copy this entire content to: Firebase Console → Firestore → Rules tab
// Then click "Publish"

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Allow users to manage their own documents
    match /users/{userId} {
      allow read, write: if request.auth.uid == userId;
    }
    
    // Waitlists collection
    match /waitlists/{waitlistId} {
      // Allow anyone to read waitlist details (for public queue page)
      allow read: if true;
      
      // Only the business owner can write/update/delete their waitlist
      allow write: if request.auth.uid == resource.data.businessId;
      allow create: if request.auth != null;
      allow delete: if request.auth.uid == resource.data.businessId;
      
      // Customers subcollection
      match /customers/{customerId} {
        // Anyone can read customers (public queue visibility)
        allow read: if true;
        
        // Only authenticated users can add/remove customers
        allow create, delete: if request.auth != null;
        
        // Anyone can update customer data (position, served status)
        allow update: if true;
      }
    }
  }
}

// EXPLANATION:

// 1. Users Collection
//    - Each user has a document with their email and role
//    - Only they can read/write their own document
//    - Field: role = "business" or "customer"

// 2. Waitlists Collection
//    - Businesses create waitlist documents
//    - businessId field matches auth UID
//    - Only business owner can write
//    - Public read allows customers to see queue info
//    - Example: /waitlists/{waitlistId}

// 3. Customers Subcollection
//    - Stored under each waitlist
//    - Contains individual customer records
//    - Fields: name, phone, position, addedAt, served
//    - Any authenticated user can read/write
//    - (App logic prevents cross-access)
//    - Path: /waitlists/{waitlistId}/customers/{customerId}

// IMPORTANT NOTES:

// • Test Mode allows all reads/writes (development only)
// • These rules are for production (more restrictive)
// • businessId stored in waitlist document
// • Role checked in React (app logic layer)
// • Customers don't need auth to join queue (public access)
